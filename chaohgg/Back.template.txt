<script>
    window.chaohggDebug = true;
    (function bootstrap(selector, attempt = 0) {
        if (typeof window.chaohggInit !== 'function') {
            if (attempt < 30) {
                requestAnimationFrame(() => bootstrap(selector, attempt + 1));
            }
            return;
        }
        const result = window.chaohggInit(selector);
        if (typeof window.chaohggReportDebug === 'function') {
            window.chaohggReportDebug('bootstrap-init', result?.error || 'success', {
                attempt,
                result
            });
        } else if (typeof console !== 'undefined' && console.log) {
            console.log('[chaohgg] init attempt', attempt, result);
        }
        if (result && !result.error) {
            if (typeof window.chaohggRevealAll === 'function') {
                window.chaohggRevealAll(result.root || document.querySelector('.chaohgg-flashcard'));
            }
            if (typeof window.chaohggOpenAllPopups === 'function') {
                window.chaohggOpenAllPopups(result.root || document.querySelector('.chaohgg-flashcard'));
            }
        }
        if ((!result || result.error === 'CHAOHGG_NO_ROOT') && attempt < 30) {
            requestAnimationFrame(() => bootstrap(selector, attempt + 1));
        }
    })('.chaohgg-flashcard');
</script>
